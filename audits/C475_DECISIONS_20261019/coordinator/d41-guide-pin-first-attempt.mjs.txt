import fs from 'node:fs';import path from 'node:path';import assert from 'node:assert/strict';import{createRequire}from'node:module';import{createHash}from'node:crypto';
const req=createRequire(process.cwd()+'/port/v2/package.json'),{build}=req('esbuild');
const B='audits/C475_DECISIONS_20261019/coordinator',scratch=B+'/node_modules/c475-draft-pin';fs.mkdirSync(scratch,{recursive:true});
const release='port/v2/apps/game/src/release-content.ts',guide='port/v2/tools/slicesmoke.mjs',sha=b=>createHash('sha256').update(b).digest('hex');
async function bullets(tag){const b=await build({stdin:{contents:"export {V2_DRAFT_RELEASE} from './apps/game/src/release-content.ts';",resolveDir:path.resolve('port/v2'),sourcefile:'c475-draft-entry.ts'},bundle:true,write:false,platform:'node',format:'cjs',logLevel:'silent'});const p=path.resolve(scratch+'/'+tag+'.cjs');assert(!fs.existsSync(p));fs.writeFileSync(p,b.outputFiles[0].contents);const rows=req(p).V2_DRAFT_RELEASE.sections.flatMap(x=>x.bullets);assert.equal(rows.length,121);return rows;}
const text=fs.readFileSync(release,'utf8'),g=fs.readFileSync(guide,'utf8'),pin=/GUIDE_DRAFT_BULLET_AUTHORITY = Object\.freeze\(\{\s+count: 121,\s+sha256: '([a-f0-9]{64})',/g,m=[...g.matchAll(pin)];assert.equal(m.length,1);
const before=await bullets('before');assert.equal(sha(JSON.stringify(before)),m[0][1]);
const old='Snake victories land the trunk on the ground and keep the painted head and jaw in shape; eels keep the upper snout and lower jaw in shape through victories, hits and fainting.';
const extra=' Painted studies keep tube worms planted as their crowns move, bend soft worms along their bodies, and give urchins, small crustaceans and nautiluses distinct body movement.';
assert.equal(text.split(old).length-1,1);assert.equal(text.split('CREATURE MOTION STUDIES').length-1,1);const at=text.indexOf(old);assert(text.lastIndexOf('CREATURE MOTION STUDIES',at)>=0);fs.writeFileSync(release+'.c475tmp',text.replace(old,old+extra));fs.renameSync(release+'.c475tmp',release);
